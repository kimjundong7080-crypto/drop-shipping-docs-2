## utils/throttle.py

```python
import time
import threading


class TokenBucket:
    def __init__(self, rate: float, capacity: float | None = None):
        self.rate = rate
        self.capacity = capacity or max(1.0, rate)
        self.tokens = self.capacity
        self.last = time.monotonic()
        self.lock = threading.Lock()

    def acquire(self, amount: float = 1.0) -> None:
        with self.lock:
            while True:
                now = time.monotonic()
                self.tokens = min(
                    self.capacity,
                    self.tokens + (now - self.last) * self.rate,
                )
                self.last = now
                if self.tokens >= amount:
                    self.tokens -= amount
                    return
                time.sleep((amount - self.tokens) / self.rate)
```

## utils/logging_setup.py

```python
import logging
import re

_MASK_KEYS = re.compile(
    r"(access[_-]?token|authorization|api[_-]?key|client[_-]?secret|"
    r"open[_-]?id|sign|password|cookie|shpat_[A-Za-z0-9]+)",
    re.IGNORECASE,
)
_EMAIL = re.compile(r"([A-Za-z0-9._%+-]+)@([A-Za-z0-9.-]+)")
_PHONE = re.compile(r"\b(\+?\d{1,3})?[\s.-]?\(?\d{2,4}\)?[\s.-]?\d{3,4}[\s.-]?\d{4}\b")


class PIIRedactionFilter(logging.Filter):
    def filter(self, record: logging.LogRecord) -> bool:
        try:
            msg = record.getMessage()
        except Exception:
            return True
        msg = _MASK_KEYS.sub(r"\1=***", msg)
        msg = _EMAIL.sub(r"\1***@\2", msg)
        msg = _PHONE.sub("***-****-****", msg)
        record.msg = msg
        record.args = ()
        return True


def setup_logging(level=logging.INFO):
    handler = logging.StreamHandler()
    handler.setFormatter(logging.Formatter(
        "%(asctime)s %(levelname)s %(name)s :: %(message)s"
    ))
    handler.addFilter(PIIRedactionFilter())
    root = logging.getLogger()
    root.handlers.clear()
    root.addHandler(handler)
    root.setLevel(level)
```

## utils/audit.py

```python
import hashlib
import json
from datetime import datetime, timezone
from db import get_conn


def _genesis_hash() -> str:
    return "0" * 64


def _last_hash() -> str:
    with get_conn() as conn:
        row = conn.execute(
            "SELECT entry_hash FROM audit_log ORDER BY id DESC LIMIT 1"
        ).fetchone()
    return row["entry_hash"] if row else _genesis_hash()


def append_audit(actor: str, action: str, target: str,
                 before: dict | None = None, after: dict | None = None,
                 request_id: str | None = None) -> str:
    prev = _last_hash()
    payload = {
        "ts": datetime.now(timezone.utc).isoformat(),
        "actor": actor,
        "action": action,
        "target": target,
        "before": before,
        "after": after,
        "request_id": request_id,
        "prev_hash": prev,
    }
    entry_hash = hashlib.sha256(
        json.dumps(payload, sort_keys=True, default=str).encode()
    ).hexdigest()
    with get_conn() as conn:
        conn.execute(
            """INSERT INTO audit_log
               (ts, actor, action, target, before_json, after_json,
                request_id, prev_hash, entry_hash)
               VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)""",
            (
                payload["ts"], actor, action, target,
                json.dumps(before, default=str) if before else None,
                json.dumps(after, default=str) if after else None,
                request_id, prev, entry_hash,
            ),
        )
    return entry_hash


def verify_audit_chain() -> bool:
    with get_conn() as conn:
        rows = conn.execute("SELECT * FROM audit_log ORDER BY id").fetchall()
    prev = _genesis_hash()
    for r in rows:
        if r["prev_hash"] != prev:
            return False
        payload = {
            "ts": r["ts"], "actor": r["actor"], "action": r["action"],
            "target": r["target"],
            "before": json.loads(r["before_json"]) if r["before_json"] else None,
            "after": json.loads(r["after_json"]) if r["after_json"] else None,
            "request_id": r["request_id"], "prev_hash": prev,
        }
        expected = hashlib.sha256(
            json.dumps(payload, sort_keys=True, default=str).encode()
        ).hexdigest()
        if expected != r["entry_hash"]:
            return False
        prev = r["entry_hash"]
    return True
```

```python
import hmac
import hashlib
import base64
from fastapi import Header, HTTPException
from config import cfg


def verify_cj(raw_body: bytes, sign_header: str) -> bool:
    if not sign_header:
        return False
    expected = base64.b64encode(
        hmac.new(cfg.cj_open_id.encode(), raw_body, hashlib.sha256).digest()
    ).decode()
    return hmac.compare_digest(sign_header, expected)


def verify_shopify(raw_body: bytes, hmac_header: str) -> bool:
    if not hmac_header:
        return False
    computed = hmac.new(
        cfg.shopify_client_secret.encode(), raw_body, hashlib.sha256
    ).digest()
    try:
        provided = base64.b64decode(hmac_header)
    except Exception:
        return False
    return hmac.compare_digest(computed, provided)


def verify_woo(raw_body: bytes, sign_header: str, secret: str) -> bool:
    if not sign_header:
        return False
    expected = base64.b64encode(
        hmac.new(secret.encode(), raw_body, hashlib.sha256).digest()
    ).decode()
    return hmac.compare_digest(sign_header, expected)


async def require_api_key(x_api_key: str = Header(default="")) -> None:
    if not x_api_key or not hmac.compare_digest(x_api_key, cfg.local_api_key):
        raise HTTPException(status_code=401, detail="invalid api key")
```

## clients/cj_client.py

```python
import json
import time
from pathlib import Path

import requests
from tenacity import (
    retry, wait_exponential, stop_after_attempt,
    retry_if_exception_type,
)

from config import cfg
from utils.throttle import TokenBucket

TOKEN_CACHE = Path(".cj_token.json")


class TransientAPIError(Exception):
    pass


class CJClient:
    def __init__(self):
        self.session = requests.Session()
        self.bucket = TokenBucket(rate=cfg.cj_qps)
        self._token: str | None = self._load_token()

    def _load_token(self) -> str | None:
        if TOKEN_CACHE.exists():
            try:
                data = json.loads(TOKEN_CACHE.read_text())
                if data.get("expiry", 0) > time.time() + 60:
                    return data["token"]
            except Exception:
                pass
        return None

    def _authenticate(self) -> str:
        resp = self.session.post(
            f"{cfg.cj_base}/authentication/getAccessToken",
            json={"email": cfg.cj_email, "password": cfg.cj_api_key},
            timeout=30,
        )
        resp.raise_for_status()
        body = resp.json()
        if not body.get("result"):
            raise RuntimeError(f"CJ auth failed: {body}")
        token = body["data"]["accessToken"]
        expiry = time.time() + body["data"].get("expiresIn", 3600) - 120
        TOKEN_CACHE.write_text(json.dumps({"token": token, "expiry": expiry}))
        return token

    @property
    def token(self) -> str:
        if self._token is None:
            self._token = self._authenticate()
        return self._token

    def _raw_get(self, path: str, params: dict | None = None) -> dict:
        self.bucket.acquire()
        resp = self.session.get(
            f"{cfg.cj_base}{path}",
            headers={"CJ-Access-Token": self.token},
            params=params or {},
            timeout=30,
        )
        if resp.status_code == 401:
            TOKEN_CACHE.unlink(missing_ok=True)
            self._token = self._authenticate()
            return self._raw_get(path, params)
        if resp.status_code in (429, 500, 502, 503):
            raise TransientAPIError(f"CJ transient {resp.status_code}")
        resp.raise_for_status()
        body = resp.json()
        if not body.get("result"):
            raise RuntimeError(f"CJ error {path}: {body}")
        return body["data"]

    @retry(
        retry=retry_if_exception_type(TransientAPIError),
        wait=wait_exponential(multiplier=1, min=2, max=60),
        stop=stop_after_attempt(5),
        reraise=True,
    )
    def get(self, path: str, params: dict | None = None) -> dict:
        return self._raw_get(path, params)

    # ---- 상품 관련 ----
    def list_products(self, keyword: str, page: int = 1, size: int = 50) -> list[dict]:
        data = self.get(
            "/product/list",
            {"productNameEn": keyword, "pageNum": page, "pageSize": size},
        )
        return data.get("list", [])

    def variant_query(self, pid: str) -> dict:
        return self.get("/product/variant/query", {"pid": pid})

    # ---- 운임 / 물류 ----
    def calculate_freight(self, payload: dict) -> dict:
        self.bucket.acquire()
        resp = self.session.post(
            f"{cfg.cj_base}/logistic/freightCalculate",
            headers={"CJ-Access-Token": self.token},
            json=payload,
            timeout=30,
        )
        resp.raise_for_status()
        return resp.json().get("data", {})

    # ---- 주문 ----
    def create_order(self, payload: dict) -> dict:
        self.bucket.acquire()
        resp = self.session.post(
            f"{cfg.cj_base}/shopping/order/createOrder",
            headers={"CJ-Access-Token": self.token},
            json=payload,
            timeout=60,
        )
        resp.raise_for_status()
        body = resp.json()
        if not body.get("result"):
            raise RuntimeError(f"CJ createOrder failed: {body}")
        return body

    def query_order(self, cj_order_id: str) -> dict:
        return self.get("/shopping/order/getOrderDetail", {"orderId": cj_order_id})
```

## clients/woo_client.py

```python
import os
import requests
from woocommerce import API

from config import cfg
from models import NormalizedProduct
from db import get_conn


wcapi = API(
    url=cfg.wc_url,
    consumer_key=cfg.wc_ck,
    consumer_secret=cfg.wc_cs,
    version="wc/v3",
    timeout=60,
)


def _chunks(seq, n):
    for i in range(0, len(seq), n):
        yield seq[i:i + n]


def _extract_attributes(variants):
    options: dict[str, set] = {}
    for v in variants:
        for k, val in v.options.items():
            options.setdefault(k, set()).add(val)
    return [
        {"name": k, "visible": True, "variation": True, "options": sorted(vals)}
        for k, vals in options.items()
    ]


def create_product(p: NormalizedProduct, canonical_sku: str) -> dict:
    attributes = _extract_attributes(p.variants)
    payload = {
        "name": p.title,
        "type": "variable" if len(p.variants) > 1 else "simple",
        "status": "draft",
        "description": p.description_html,
        "images": [{"src": u} for u in p.images],
        "attributes": attributes,
    }
    if len(p.variants) == 1:
        v = p.variants[0]
        payload.update({
            "regular_price": str(v.price),
            "sku": v.supplier_sku,
            "manage_stock": True,
            "stock_quantity": v.stock,
        })

    resp = wcapi.post("products", payload)
    resp.raise_for_status()
    product = resp.json()

    if len(p.variants) > 1:
        _create_variations(product["id"], p.variants, attributes, canonical_sku)
    else:
        _record_channel_variant(
            canonical_sku, "woocommerce",
            str(product["id"]), str(product["id"]),
            product.get("sku"), p.variants[0].stock,
        )

    return product


def _create_variations(product_id: int, variants, attributes, canonical_sku: str):
    attr_names = [a["name"] for a in attributes]
    items = [{
        "regular_price": str(v.price),
        "sku": v.supplier_sku,
        "manage_stock": True,
        "stock_quantity": v.stock,
        "attributes": [
            {"name": name, "option": v.options[name]}
            for name in attr_names if name in v.options
        ],
    } for v in variants]

    for chunk in _chunks(items, 100):
        resp = wcapi.post(f"products/{product_id}/variations/batch", {"create": chunk})
        resp.raise_for_status()
        created = resp.json().get("create", [])
        for c in created:
            _record_channel_variant(
                canonical_sku, "woocommerce", str(product_id), str(c["id"]),
                c.get("sku"), 0,
            )


def _record_channel_variant(canonical_sku, channel, product_id, variant_id, sku, qty):
    with get_conn() as conn:
        conn.execute(
            """INSERT OR IGNORE INTO channel_variants
               (canonical_sku, channel, channel_product_id,
                channel_variant_id, channel_sku, last_pushed_qty)
               VALUES (?, ?, ?, ?, ?, ?)""",
            (canonical_sku, channel, product_id, variant_id, sku, qty),
        )


def batch_update_variations(product_id: int, updates: list[dict]) -> None:
```

```python
import time
import requests

from config import cfg


class ShopifyClient:
    def __init__(self):
        self.url = (
            f"https://{cfg.shopify_shop}/admin/api/"
            f"{cfg.shopify_api_version}/graphql.json"
        )
        self.session = requests.Session()
        self.session.headers.update({
            "X-Shopify-Access-Token": cfg.shopify_token,
            "Content-Type": "application/json",
        })
        self.available = 1000
        self.restore = 100

    def execute(self, query: str, variables: dict | None = None) -> dict:
        for _ in range(5):
            resp = self.session.post(
                self.url,
                json={"query": query, "variables": variables or {}},
                timeout=60,
            )
            if resp.status_code == 429:
                time.sleep(2)
                continue
            resp.raise_for_status()
            body = resp.json()
            cost = body.get("extensions", {}).get("cost", {})
            if cost:
                ts = cost.get("throttleStatus", {})
                self.available = ts.get("currentlyAvailable", self.available)
                self.restore = ts.get("restoreRate", self.restore)
            if body.get("errors"):
                raise RuntimeError(body["errors"])
            return body["data"]
        raise RuntimeError("Shopify retries exhausted")

    def wait_for_budget(self, need: int):
        while self.available < need:
            time.sleep(1)
            self.available = min(1000, self.available + self.restore)

    if errs:

            "name": "available",

def fulfill_order(client: ShopifyClient, order_gid: str,
                  tracking_number: str, tracking_url: str, carrier: str):
    fo = client.execute(FULFILLMENT_ORDERS, {"orderId": order_gid})
    fos = [e["node"] for e in fo["order"]["fulfillmentOrders"]["edges"]]
    if not fos:
        raise RuntimeError("no fulfillment orders")

    line_items = []
    for f in fos:
        for li in f["lineItems"]["edges"]:
            node = li["node"]
            if node["remainingQuantity"] <= 0:
                continue
            line_items.append({
                "fulfillmentOrderId": f["id"],
                "fulfillmentOrderLineItems": [{
                    "id": node["id"],
                    "quantity": node["remainingQuantity"],
                }],
            })
    if not line_items:
        raise RuntimeError("nothing to fulfill")

    d = client.execute(FULFILLMENT_CREATE, {
        "fulfillment": {
            "lineItemsByFulfillmentOrder": line_items,
            "trackingInfo": {
                "number": tracking_number,
                "url": tracking_url,
                "company": carrier,
            },
            "notifyCustomer": False,
        }
    })
    errs = d["fulfillmentCreate"]["userErrors"]
    if errs:
        raise RuntimeError(errs)
    return d["fulfillmentCreate"]["fulfillment"]
```

```python
from decimal import Decimal, ROUND_HALF_UP
from config import cfg


def compute_sell_price(supplier_price: Decimal, freight: Decimal = Decimal("0")) -> Decimal:
    base = supplier_price + freight
    with_margin = base * (Decimal("1") + cfg.pricing_margin)
    # .99 로 끝나는 반올림
    integer = with_margin.to_integral_value(rounding=ROUND_HALF_UP)
    candidate = (integer - Decimal("0.01")).quantize(Decimal("0.01"))
    if candidate < with_margin:
        candidate = integer.quantize(Decimal("0.01"))
    return candidate.quantize(Decimal("0.01"))


def assert_positive_margin(sell: Decimal, cost: Decimal) -> None:
    if sell <= cost:
        raise ValueError(f"negative margin: sell={sell} cost={cost}")
```

```python
import logging
from clients.cj_client import CJClient
from clients import woo_client, shopify_client
```

```python
log = logging.getLogger(__name__)
shopify = shopify_client.ShopifyClient()


def sync_inventory(stale_minutes: int = 30) -> dict:
    cj = CJClient()
    summary = {"updated": 0, "errors": 0}

    with get_conn() as conn:
        rows = conn.execute("""
            SELECT cv.*, sp.supplier_vid, sp.supplier_sku, sp.supplier_pid
            FROM channel_variants cv
            JOIN mapping m ON m.canonical_sku = cv.canonical_sku
            JOIN supplier_products sp ON sp.supplier_sku = m.supplier_sku
        """).fetchall()

    by_pid: dict[str, list] = {}
    for r in rows:
        by_pid.setdefault(r["supplier_pid"], []).append(r)

    for pid, variants in by_pid.items():
        try:
            data = cj.variant_query(pid)
        except Exception as e:
            log.error("variant_query %s failed: %s", pid, e)
            summary["errors"] += 1
            continue

        stock_by_sku = {
            v["variantSku"]: int(v.get("variantInventory", 0) or 0)
            for v in data.get("variants", [])
        }

        woo_by_product: dict[int, list] = {}
        for r in variants:
            new_qty = stock_by_sku.get(r["supplier_sku"])
            if new_qty is None:
                continue
            if r["channel"] == "woocommerce":
                woo_by_product.setdefault(int(r["channel_product_id"]), []).append(
                    {"id": int(r["channel_variant_id"]),
                     "manage_stock": True,
                     "stock_quantity": new_qty}
                )
            elif r["channel"] == "shopify":
                delta = new_qty - (r["last_pushed_qty"] or 0)
                try:
                    shopify_client.adjust_inventory(
                        shopify, r["inventory_item_id"], delta
                    )
                    _save_pushed(r["id"], new_qty)
                    summary["updated"] += 1
                except Exception as e:
                    log.error("shopify inventory %s: %s", r["id"], e)
                    summary["errors"] += 1

        for product_id, updates in woo_by_product.items():
            try:
                woo_client.batch_update_variations(product_id, updates)
                for u in updates:
                    _save_pushed_by_variant("woocommerce", str(u["id"]), u["stock_quantity"])
                summary["updated"] += len(updates)
            except Exception as e:
                log.error("woo batch %s: %s", product_id, e)
                summary["errors"] += 1

    return summary
```

```python
        "logisticName": order_row.get("preferred_logistic") or "CJPacket Ordinary",

        "remark": f"{order_row['channel']} / {order_row['channel_order_num']}",
        "isSandbox": is_sandbox,
        "products": products,
    }


def sandbox_preview(order_id: int) -> dict:
    p = preview_order(order_id)
    cj = CJClient()
    payload = _build_cj_payload(p["order"], p["items"], is_sandbox=1)
    resp = cj.session.post(
        f"{cj_base()}/shopping/order/createOrder",
        headers={"CJ-Access-Token": cj.token},
        json=payload,
        timeout=60,
    )
    resp.raise_for_status()
    return resp.json()


def cj_base() -> str:
    from config import cfg
    return cfg.cj_base


def place_order(order_id: int) -> dict:
    # 1) PENDING 선기록
    with get_conn() as conn:
        try:
            conn.execute(
                """INSERT INTO cj_orders (local_order_id, cj_order_number, status)
                   VALUES (?, ?, 'PENDING')""",
                (order_id, f"PENDING-{order_id}"),
            )
        except sqlite3.IntegrityError:
            row = conn.execute(
                "SELECT * FROM cj_orders WHERE local_order_id = ?", (order_id,)
            ).fetchone()
            return {"skipped": "already_exists", "status": row["status"] if row else None}

    p = preview_order(order_id)
    payload = _build_cj_payload(p["order"], p["items"], is_sandbox=0)

    cj = CJClient()
    result = cj.create_order(payload)

    cj_order_id = str(result.get("data"))
    with get_conn() as conn:
        conn.execute(
            """UPDATE cj_orders
               SET cj_order_id = ?, cj_order_number = ?, status = 'CREATED',
                   updated_at = CURRENT_TIMESTAMP
               WHERE local_order_id = ?""",
            (cj_order_id, payload["orderNumber"], order_id),
        )

    append_audit("system", "cj.order.create", f"order:{order_id}",
                 after={"cj_order_id": cj_order_id})

    if p["order"]["channel"] == "woocommerce":
        woo_client.push_order_note(int(p["order"]["channel_order_id"]),
                                   f"CJ order {cj_order_id} created")
        woo_client.update_order_status(int(p["order"]["channel_order_id"]), "processing")

    return {"cj_order_id": cj_order_id, "status": "CREATED"}


def push_tracking(order_id: int) -> dict:
    with get_conn() as conn:
        row = conn.execute(
            "SELECT * FROM cj_orders WHERE local_order_id = ?", (order_id,)
        ).fetchone()
        order = conn.execute(
            "SELECT * FROM orders WHERE id = ?", (order_id,)
        ).fetchone()
    if not row or not row["cj_order_id"]:
        raise ValueError("no cj order")

    cj = CJClient()
    detail = cj.query_order(row["cj_order_id"])
    tracking = detail.get("trackNumber") or detail.get("trackingNumber")
    carrier = detail.get("logisticName") or row["logistic_name"] or "CJ"

    with get_conn() as conn:
        conn.execute(
            """UPDATE cj_orders SET tracking_number = ?, logistic_name = ?,
               status = 'SHIPPED', updated_at = CURRENT_TIMESTAMP
               WHERE local_order_id = ?""",
            (tracking, carrier, order_id),
        )

    if not tracking:
        return {"status": "no_tracking"}

    if order["channel"] == "shopify":
        shopify_client.fulfill_order(
            shopify, order["channel_order_id"], tracking,
            f"https://t.17track.net/en#nums={tracking}", carrier,
        )
    else:
        woo_client.push_order_note(int(order["channel_order_id"]),
                                   f"Tracking: {tracking} ({carrier})")

    append_audit("system", "order.tracking.push", f"order:{order_id}",
                 after={"tracking": tracking, "carrier": carrier})
    return {"tracking": tracking, "carrier": carrier}
```

```python
from clients import woo_client, shopify_client
from db import get_conn

shopify = shopify_client.ShopifyClient()


def set_listing(canonical_sku: str, channels: list[str], visible: bool) -> dict:
    out = {}
    with get_conn() as conn:
        rows = conn.execute(
            "SELECT * FROM channel_variants WHERE canonical_sku = ?", (canonical_sku,)
        ).fetchall()
    if not rows:
        raise ValueError(f"no channel variants for {canonical_sku}")

    publication_ids = None
    if "shopify" in channels:
        publication_ids = [p["id"] for p in shopify_client.list_publications(shopify)]

    seen = set()
    for r in rows:
        if r["channel"] not in channels:
            continue
        key = (r["channel"], r["channel_product_id"])
        if key in seen:
            continue
        seen.add(key)
        try:
            if r["channel"] == "woocommerce":
                woo_client.set_listing_visibility(int(r["channel_product_id"]), visible)
            elif r["channel"] == "shopify":
                shopify_client.set_listing(
                    shopify, r["channel_product_id"], publication_ids, visible
                )
            out[f"{r['channel']}:{r['channel_product_id']}"] = "ok"
            append_audit("system", "listing.set", f"{r['channel']}:{canonical_sku}",
                         after={"visible": visible})
        except Exception as e:
            out[f"{r['channel']}:{r['channel_product_id']}"] = f"error: {e}"
    return out
```

## webhook/receiver.py

```python
import hashlib
import json
import logging

from fastapi import FastAPI, Request, HTTPException

from db import get_conn
from utils.security import verify_cj, verify_shopify

log = logging.getLogger(__name__)
app = FastAPI(title="Dropship Webhook Receiver", docs_url=None, redoc_url=None)


def _record(provider: str, delivery_id: str, topic: str | None) -> bool:
    """중복 배송 차단. 새 이벤트면 True."""
    try:
        with get_conn() as conn:
            conn.execute(
                """INSERT INTO webhook_events (provider, delivery_id, topic)
                   VALUES (?, ?, ?)""",
                (provider, delivery_id, topic),
            )
        return True
    except Exception:
        return False


@app.post("/webhooks/cj")
async def cj_webhook(request: Request):
    raw = await request.body()
    sign = request.headers.get("sign", "")
    if not verify_cj(raw, sign):
        raise HTTPException(401, "invalid signature")
    try:
        body = json.loads(raw)
    except json.JSONDecodeError:
        raise HTTPException(400, "invalid json")

    delivery_id = hashlib.sha256(raw).hexdigest()
    topic = body.get("type") or body.get("topic")
    if not _record("cj", delivery_id, topic):
        return {"ok": True, "duplicate": True}

    # 백그라운드 워커로 위임 (여기서는 로그만)
    log.info("cj webhook %s received", topic)
    return {"ok": True}


@app.post("/webhooks/shopify")
async def shopify_webhook(request: Request):
    raw = await request.body()
    hmac_header = request.headers.get("X-Shopify-Hmac-SHA256", "")
    if not verify_shopify(raw, hmac_header):
        raise HTTPException(401, "invalid signature")

    delivery_id = request.headers.get("X-Shopify-Webhook-Id", "")
    if not delivery_id:
        delivery_id = hashlib.sha256(raw).hexdigest()
    topic = request.headers.get("X-Shopify-Topic")
    if not _record("shopify", delivery_id, topic):
        return {"ok": True, "duplicate": True}

    log.info("shopify webhook %s received", topic)
    return {"ok": True}
```

## 이 코드에서 의도적으로 비워 둔 부분

- **CJ 웹훅 등록(`webhook/set`)**: 계정 등급·7개 토픽 상태 관리가 필요하므로 `CJClient`에 메서드만 추가하고 호출은 수동으로 두었습니다. 터널 주소가 확정된 뒤 `product`·`stock`·`order`·`logistics` 4개를 **항상 함께** 전송하도록 구현하세요.
- **Shopify 리뷰 어댑터**: `comment_service.send_reply`가 `woocommerce`만 처리합니다. Judge.me 등 리뷰 앱의 API 키를 `.env`에 추가하고 `ShopifyReviewProvider`를 구현하면 됩니다.
- **이미지 재업로드**: `utils/images.py`를 만들고 `create_product` 진입부에서 공급처 이미지를 로컬로 받아 채널 미디어 API로 업로드하는 로직을 추가하세요.
- **`openId` 마스킹**: `logging_setup.py`의 정규식이 `open_id`/`openId` 키를 마스킹하지만, `webhook/receiver.py`에서 예외 스택을 그대로 로그에 남기지 않도록 `log.exception` 대신 커스텀 핸들러를 두는 편이 더 안전합니다.

이 상태로 `python main.py doctor`가 통과하면, 상품 동기화부터 발주·리스팅·댓글까지 로컬에서 완결되는 운영 콘솔이 동작합니다.
